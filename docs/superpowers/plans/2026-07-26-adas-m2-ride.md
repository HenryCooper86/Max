# ADAS M2 — On-Bike Runtime Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Run the M1 pipeline on the bike, logging every alert it would fire, with alert audio built and shipped locked.

**Architecture:** A runtime thread per camera inside the vision sidecar drives M1's pure modules unchanged and emits hazard events over the existing NDJSON channel. The agent logs them; when `MAX_ADAS_AUDIO=1` it also preempts Max and plays a pre-rendered WAV through the normal playback path, so the alert reaches the AEC reference like any other audio.

**Tech Stack:** Python 3.10 + numpy on the board, Node 22 / TypeScript in the agent, vitest, pytest, ffmpeg.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-07-26-adas-m2-ride-design.md`. Parents:
  `2026-07-25-embedded-adas-bpu-design.md`, `2026-07-26-adas-m1-measure-design.md`.
- **`MAX_ADAS_AUDIO` defaults to `0`.** Shadow is the default; the agent stays silent.
- **M1's constants stay untuned.** Nothing in M2 changes a threshold.
- **Alert audio goes through `playback.write()`**, never a separate `aplay`. That is
  what feeds the AEC reference, the soft limiter and the pacing.
- **An alert cannot be preempted.** `playback.flush()` is a no-op while one plays.
- **`hazard.report` flows upward only.** The relay must never echo it as `hazard.alert`.
- **Shadow mode requires recording.** No alerts without evidence on disk.
- Python tests run on the board: `ssh root@192.168.50.9 'cd /opt/max/rdk/vision && python3 -m pytest tests/ -q'`.
  TypeScript tests run locally: `npx vitest run`. The full gate is `npm run check`.
- Follow existing conventions: `from __future__ import annotations`, frozen dataclasses,
  docstrings that say _why_; Python tests start with
  `sys.path.insert(0, str(Path(__file__).resolve().parents[1]))`.

```ts
/**
 * Plays a pre-rendered hazard clip with absolute priority.
 *
 * Audio goes through `playback.write`, never a separate aplay. That path
 * already carries the soft limiter, the pacing, and — critically — the AEC
 * reference tap. An alert leaves the same speaker and enters the same
 * microphone as Max's voice, so if APM is not told about it, the mic hears an
 * uncancelled warning, Doubao's VAD fires on it, and the echo loop returns:
 * triggered by our own safety alert.
 */

import { existsSync, readFileSync } from "node:fs";
import { join } from "node:path";

/** playback.write consumes 24 kHz PCM16, so clips must be rendered at that rate. */
export const ALERT_RATE = 24_000;

/** 20 ms at 24 kHz. Matches the cadence the relay delivers speech in. */
const CHUNK_SAMPLES = 480;

export type AlertPlayback = {
  write(pcm: Int16Array): void;
  flush(): void;
  holdPriority(): void;
  releasePriority(): void;
};

export type AlertPlayerOptions = {
  dir: string;
  playback: AlertPlayback;
  now?: () => number;
};

export function readWavMono24k(path: string): Int16Array {
  const buffer = readFileSync(path);
  const channels = buffer.readUInt16LE(22);
  const rate = buffer.readUInt32LE(24);
  const bits = buffer.readUInt16LE(34);
  if (rate !== ALERT_RATE) {
    throw new Error(`${path}: expected ${ALERT_RATE} Hz, got ${rate}`);
  }
  if (channels !== 1 || bits !== 16) {
    throw new Error(`${path}: expected mono 16-bit, got ${channels}ch ${bits}-bit`);
  }
  const bytes = buffer.subarray(44);
  const samples = new Int16Array(bytes.length >> 1);
  for (let i = 0; i < samples.length; i += 1) samples[i] = bytes.readInt16LE(i * 2);
  return samples;
}

export function createAlertPlayer(options: AlertPlayerOptions): {
  play(threat: string): Promise<number | null>;
} {
  const now = options.now ?? (() => Date.now());

  return {
    async play(threat: string): Promise<number | null> {
      const path = join(options.dir, `${threat}.wav`);
      if (!existsSync(path)) {
        console.log(`[adas] no clip for ${threat}; staying silent`);
        return null;
      }

      let pcm: Int16Array;
      try {
        pcm = readWavMono24k(path);
      } catch (error) {
        console.log(`[adas] unusable clip for ${threat}: ${(error as Error).message}`);
        return null;
      }

      const started = now();
      // Preempt first, then take the lock: flushing after holding would be a
      // no-op against our own lock and Max would talk over the warning.
      options.playback.flush();
      options.playback.holdPriority();
      try {
        for (let offset = 0; offset < pcm.length; offset += CHUNK_SAMPLES) {
          options.playback.write(pcm.subarray(offset, offset + CHUNK_SAMPLES));
        }
      } finally {
        options.playback.releasePriority();
      }
      return now() - started;
    }
  };
}
```

```ts
/**
 * Turns a sidecar hazard event into either a log line or a spoken warning.
 *
 * Shadow mode is the default and does neither audio nor reporting: the board
 * made no sound, so there is no real alert for the transcript, and a shadow run
 * that reported would be indistinguishable from an armed one in the QA record.
 */

export type HazardEvent = {
  event: string;
  threat?: string;
  direction?: string;
  severity?: string;
  label?: string;
  message?: string;
  advice?: string;
  ttcSeconds?: number;
};

export type HazardHandlerOptions = {
  /** config.adasAudio. False until the M1 gate is met. */
  armed: boolean;
  player: { play(threat: string): Promise<number | null> };
  report(event: Record<string, unknown>): void;
  log?: (line: string) => void;
};

export function createHazardHandler(options: HazardHandlerOptions): {
  handle(event: HazardEvent): Promise<void>;
} {
  const log = options.log ?? ((line: string) => console.log(line));

  return {
    async handle(event: HazardEvent): Promise<void> {
      if (event.event !== "hazard" || !event.threat) return;

      if (!options.armed) {
        log(`[adas] shadow: ${event.threat} ttc=${event.ttcSeconds ?? "n/a"}`);
        return;
      }

      const latencyMs = await options.player.play(event.threat);

      try {
        options.report({
          type: "hazard.report",
          threat: event.threat,
          direction: event.direction,
          severity: event.severity,
          label: event.label,
          message: event.message,
          advice: event.advice,
          ttcSeconds: event.ttcSeconds,
          latencyMs
        });
      } catch (error) {
        // Reporting is best-effort and must never cost the rider the warning.
        log(`[adas] hazard.report failed: ${(error as Error).message}`);
      }
    }
  };
}
```

```python
"""The shadow-mode alert log — M2's actual deliverable.

Every alert the pipeline would have fired, with enough context to explain
itself without re-running anything. The ride is over by the time anyone reads
this, so a record that only says "forward_collision at 12:04" is not evidence,
it is a rumour.

Records are appended and flushed one at a time. A ride ends when the bike's
power does, and a buffered writer would lose exactly the tail worth reviewing.
"""

from __future__ import annotations

import json
from dataclasses import asdict, dataclass
from pathlib import Path


@dataclass(frozen=True)
class ShadowRecord:
    threat_id: str
    severity: str
    camera: str
    #: Pipeline time, matching the frame timestamps.
    t_ms: int
    #: Wall clock, for lining a ride up against a transcript.
    wall_ms: int
    ttc_s: float | None

    #: Provenance — which recording, and where in it.
    ride_id: str
    segment: str
    segment_offset_ms: int

    #: The pipeline's own view, so a false alarm can be diagnosed offline.
    track_id: int
    box_height: float
    closing_rate: float | None
    track_frames: int
    ego_moving: bool

    #: Populated only when armed.
    latency_ms: float | None = None


class ShadowLog:
    def __init__(self, path: Path | str) -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        # Append, never truncate: a restart mid-ride must not erase the alerts
        # that already fired.
        self._fh = self.path.open("a", encoding="utf-8")

    def append(self, record: ShadowRecord) -> None:
        self._fh.write(json.dumps(asdict(record), ensure_ascii=False) + "\n")
        self._fh.flush()

    def close(self) -> None:
        if not self._fh.closed:
            self._fh.close()

    def __enter__(self) -> "ShadowLog":
        return self

    def __exit__(self, *_exc) -> None:
        self.close()


def read_shadow(path: Path | str) -> list[ShadowRecord]:
    lines = Path(path).read_text(encoding="utf-8").strip()
    if not lines:
        return []
    return [ShadowRecord(**json.loads(line)) for line in lines.split("\n")]
```

```python
"""The on-bike loop. A thin driver around M1's pure modules.

It owns cameras, timing and bookkeeping; every decision belongs to threat.py
and suppress.py, imported unchanged. That is what makes a threshold tuned
against recorded rides the same code that fires on the road.
"""

from __future__ import annotations

import time
from dataclasses import dataclass
from typing import Callable

from perception.tracker import Tracker
from perception.types import Detection, EgoState

from .catalog import CATALOG
from .ego_motion import estimate
from .shadow import ShadowLog, ShadowRecord
from .suppress import Suppressor
from .threat import evaluate
from .ttc import height_rate


class RecordingRequired(RuntimeError):
    """Shadow mode without a recorder produces alerts nobody can review."""


@dataclass
class RuntimeConfig:
    ride_id: str
    camera: str
    #: Shadow mode is meaningless without clips to review; see the M2 spec.
    recording: bool = True
    #: Injected in tests; None means use the BPU detector passed to run_camera.
    detect: Callable[[object, int], list[Detection]] | None = None
    segment_pattern: str = "{camera}_{index:06d}.h264"
    segment_seconds: float = 60.0


def run_camera(
    source,
    config: RuntimeConfig,
    shadow: ShadowLog,
    emit: Callable[[dict], None],
    detector=None,
) -> int:
    """Drive one camera until its frames run out. Returns alerts fired."""
    if not config.recording:
        raise RecordingRequired(
            "shadow mode needs the recorder running: an unreviewable "
            "false-alarm rate is worse than none"
        )

    detect = config.detect
    if detect is None:
        if detector is None:
            raise ValueError("run_camera needs either config.detect or a detector")

        def detect(frame, _index):
            return detector.infer(frame.nv12, frame.width, frame.height)

    tracker = Tracker()
    suppressor = Suppressor()
    previous_nv12: bytes | None = None
    fired = 0

    for index, frame in enumerate(source.frames()):
        dets = detect(frame, index)

        if previous_nv12 is None:
            ego = EgoState(moving=False, speed_scale=0.0)
        else:
            ego = estimate(previous_nv12, frame.nv12, frame.width, frame.height)
        previous_nv12 = frame.nv12

        tracks = tracker.update(dets, frame.t_ms)
        by_id = {t.track_id: t for t in tracks}
        candidates = evaluate(tracks, ego, frame.t_ms, config.camera)
        admitted = suppressor.admit(candidates, {t.track_id: t.frames for t in tracks}, frame.t_ms)

        for alert in admitted:
            track = by_id.get(alert.track_id)
            segment_index = int(frame.t_ms / 1000.0 / config.segment_seconds)
            shadow.append(
                ShadowRecord(
                    threat_id=alert.threat_id,
                    severity=alert.severity,
                    camera=alert.camera,
                    t_ms=alert.t_ms,
                    wall_ms=int(time.time() * 1000),
                    ttc_s=alert.ttc_s,
                    ride_id=config.ride_id,
                    segment=config.segment_pattern.format(
                        camera=config.camera, index=segment_index
                    ),
                    segment_offset_ms=int(frame.t_ms - segment_index * config.segment_seconds * 1000),
                    track_id=alert.track_id,
                    box_height=track.last.height if track else 0.0,
                    closing_rate=height_rate(track) if track else None,
                    track_frames=track.frames if track else 0,
                    ego_moving=ego.moving,
                )
            )
            entry = CATALOG.get(alert.threat_id, {})
            emit(
                {
                    "event": "hazard",
                    "threat": alert.threat_id,
                    "severity": alert.severity,
                    "direction": entry.get("direction", alert.camera),
                    "label": entry.get("label", alert.threat_id),
                    "message": entry.get("message", ""),
                    "advice": entry.get("advice", ""),
                    "ttcSeconds": alert.ttc_s,
                }
            )
            fired += 1

    return fired
```

```python
"""Cut the clip around a shadow alert out of the ride recording.

Shadow mode stores a timestamp, not frames. The recorder is already writing
continuous segments, so a rolling in-memory buffer would spend ~310 MB per
camera duplicating what is on disk — on a board whose memory and thermal
headroom are both tight.

The clips this produces are the same H.264 the corpus is made of, so reviewing
a false alarm and growing the tuning corpus are the same action.

    python3 tools/clip.py shadow.jsonl --index 0 --segments /var/lib/max/corpus/ride-001
"""

from __future__ import annotations

import argparse
import subprocess
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))

from adas.shadow import read_shadow

BEFORE_S = 5.0
AFTER_S = 3.0


def clip_window(offset_ms: int, before_s: float = BEFORE_S, after_s: float = AFTER_S) -> tuple[float, float]:
    """Return (start_seconds, duration_seconds), clamped to the segment start."""
    centre = offset_ms / 1000.0
    start = max(0.0, centre - before_s)
    duration = (centre - start) + after_s
    return start, duration


def extract(
    segment_path: Path | str,
    offset_ms: int,
    out_path: Path | str,
    before_s: float = BEFORE_S,
    after_s: float = AFTER_S,
) -> Path:
    segment = Path(segment_path)
    if not segment.exists():
        raise FileNotFoundError(segment)
    start, duration = clip_window(offset_ms, before_s, after_s)
    out = Path(out_path)
    out.parent.mkdir(parents=True, exist_ok=True)
    subprocess.run(
        ["ffmpeg", "-v", "error", "-ss", f"{start:.3f}", "-i", str(segment),
         "-t", f"{duration:.3f}", "-c:v", "libx264", "-pix_fmt", "yuv420p",
         "-y", str(out)],
        check=True,
    )
    return out


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("shadow")
    parser.add_argument("--index", type=int, required=True)
    parser.add_argument("--segments", required=True)
    parser.add_argument("--out", default="/tmp/alert.mp4")
    args = parser.parse_args()

    records = read_shadow(args.shadow)
    if not 0 <= args.index < len(records):
        raise SystemExit(f"index {args.index} out of range; the log holds {len(records)}")
    record = records[args.index]
    out = extract(Path(args.segments) / record.segment, record.segment_offset_ms, args.out)
    print(f"{record.threat_id} @ {record.t_ms} ms -> {out}")


if __name__ == "__main__":
    main()
```

```python
"""Render the alert clips once, through Doubao TTS.

Pre-rendered rather than synthesised on demand: zero synthesis latency against a
2-second TTC budget, identical wording every time so the rider learns the sound,
and no network dependency at all. An ADAS alert that needs WiFi is not an ADAS
alert.

Direct TTS rather than capturing a realtime session, because the model
paraphrases — and a sound the rider is supposed to recognise cannot drift
between renders.

    DOUBAO_TTS_TOKEN=... python3 tools/render_alerts.py --out /opt/max/rdk/audio/alerts
"""

from __future__ import annotations

import argparse
import json
import os
import struct
import sys
import urllib.request
from pathlib import Path
from typing import Callable

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))

from adas.catalog import CATALOG

RATE = 24_000
ENDPOINT = "https://openspeech.bytedance.com/api/v1/tts"


def wav_bytes(pcm: bytes, rate: int = RATE) -> bytes:
    """Wrap raw mono PCM16 in a WAV header alert-player.ts will accept."""
    header = b"RIFF" + struct.pack("<I", 36 + len(pcm)) + b"WAVE"
    header += b"fmt " + struct.pack("<IHHIIHH", 16, 1, 1, rate, rate * 2, 2, 16)
    header += b"data" + struct.pack("<I", len(pcm))
    return header + pcm


def render(synth: Callable[[str], bytes], out_dir: Path | str) -> list[Path]:
    """Write one WAV per catalog entry. `synth` returns raw PCM16 at RATE."""
    out = Path(out_dir)
    out.mkdir(parents=True, exist_ok=True)
    written: list[Path] = []
    for threat, entry in CATALOG.items():
        pcm = synth(entry["message"])
        path = out / f"{threat}.wav"
        path.write_bytes(wav_bytes(pcm))
        written.append(path)
    return written


def doubao_synth(text: str) -> bytes:
    """Call Doubao TTS and return raw PCM16 at RATE.

    Credentials are separate from the relay's realtime-session ones; the relay
    only ever uses TTS inside a session.
    """
    token = os.environ.get("DOUBAO_TTS_TOKEN")
    appid = os.environ.get("DOUBAO_TTS_APPID")
    voice = os.environ.get("DOUBAO_TTS_VOICE", "zh_female_qingxin")
    if not token or not appid:
        raise SystemExit("set DOUBAO_TTS_TOKEN and DOUBAO_TTS_APPID")

    payload = {
        "app": {"appid": appid, "token": token, "cluster": "volcano_tts"},
        "user": {"uid": "max-adas"},
        "audio": {"voice_type": voice, "encoding": "pcm", "rate": RATE},
        "request": {"reqid": text[:16] or "alert", "text": text, "operation": "query"},
    }
    request = urllib.request.Request(
        ENDPOINT,
        data=json.dumps(payload).encode(),
        headers={"Authorization": f"Bearer;{token}", "Content-Type": "application/json"},
    )
    with urllib.request.urlopen(request, timeout=30) as response:
        body = json.loads(response.read().decode())
    import base64

    if "data" not in body:
        raise SystemExit(f"TTS failed: {body}")
    return base64.b64decode(body["data"])


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", default="/opt/max/rdk/audio/alerts")
    args = parser.parse_args()
    for path in render(doubao_synth, args.out):
        print(f"wrote {path} ({path.stat().st_size} bytes)")


if __name__ == "__main__":
    main()
```

```python
import threading
from pathlib import Path

from adas.runtime import RuntimeConfig, run_camera
from adas.shadow import ShadowLog


def start_adas(config: RuntimeConfig, shadow_path, emit, source_factory) -> threading.Thread:
    """Run one camera's ADAS loop on a daemon thread.

    Exceptions are caught and logged rather than raised. ADAS failing must cost
    hazard detection and nothing else — the sidecar keeps serving preview and
    gallery, and voice is a different process that this cannot reach.
    """

    def body() -> None:
        try:
            with ShadowLog(Path(shadow_path)) as shadow:
                fired = run_camera(source_factory(), config, shadow, emit)
                print(f"[adas] {config.camera}: {fired} alerts", flush=True)
        except Exception as error:  # noqa: BLE001 — see the docstring
            print(f"[adas] {config.camera} stopped: {error}", flush=True)

    thread = threading.Thread(target=body, name=f"adas-{config.camera}", daemon=True)
    thread.start()
    return thread
```
